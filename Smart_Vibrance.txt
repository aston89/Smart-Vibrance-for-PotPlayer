/*
    Smart Vibrance V1.0

    DESCRIPTION
    A lightweight, per-pixel saturation enhancement shader.
    The shader increases chroma while preserving the original pixel
    luminance. Near-neutral colors receive a reduced boost, while
    increasingly saturated colors receive progressively less
    enhancement through a chroma-dependent rolloff.

    FEATURES
    - Luminance-preserving chroma adjustment.
    - Reduced saturation boost for near-grayscale colors.
    - Progressive rolloff to avoid boosting already saturated colors.
    - Original input alpha is preserved.
    - Stateless processing: no temporal history or frame-to-frame
      stabilization is used.

    PARAMETERS
    Saturation multiplier:
        1.0 = no saturation adjustment.
        2.0 = nominal 100% increase before the protection and rolloff
              factors are applied.

    Grayscale threshold:
        Defines how close a pixel must be to neutral gray to receive
        the reduced saturation boost.

    Chroma rolloff:
        The enhancement decreases as chroma magnitude approaches 0.5,
        reaching zero at or above that level.

    NOTES
    Thresholds and chroma measurements assume RGB values in the
    normalized 0.0-1.0 range.
*/


sampler s0 : register(s0);


float3 AdjustSaturation(float3 color, float saturation)
{
    // Estimate perceptual luminance using Rec. 709 coefficients.
    float luminance = dot(color, float3(0.2126, 0.7152, 0.0722));

    // Identify pixels whose RGB channels are close to their luminance.
    // These pixels are treated as near-neutral colors.
    const float grayscaleThreshold = 0.05;

    bool isGray =
        abs(color.r - luminance) < grayscaleThreshold &&
        abs(color.g - luminance) < grayscaleThreshold &&
        abs(color.b - luminance) < grayscaleThreshold;

    // Express chroma as the RGB deviation from the neutral gray
    // corresponding to the pixel's original luminance.
    float3 chromaticity = color - luminance;

    // Measure the magnitude of that chroma deviation.
    float chromaMagnitude = length(chromaticity);

    // Reduce the enhancement for near-neutral pixels.
    // The boost is attenuated, not completely disabled.
    float scalingFactor;

    if (isGray)
    {
        scalingFactor = 0.25;
    }
    else
    {
        // Linearly reduce the enhancement as chroma increases.
        // The factor reaches zero when chromaMagnitude reaches 0.5.
        scalingFactor = 1.0 - saturate(chromaMagnitude / 0.5);
    }

    // Scale the original chroma without changing its direction.
    // At saturation = 1.0, the original chroma is unchanged.
    float3 adjustedChromaticity =
        chromaticity +
        chromaticity * (saturation - 1.0) * scalingFactor;

    // Recombine adjusted chroma with the original luminance estimate.
    float3 adjustedColor = luminance + adjustedChromaticity;

    return adjustedColor;
}


float4 main(float2 texCoord : TEXCOORD) : SV_TARGET
{
    // Read the source pixel.
    float4 inputColor = tex2D(s0, texCoord);

    // Apply saturation enhancement with near-neutral protection
    // and a gradual rolloff for increasingly saturated pixels.
    // 2.0 is the nominal +100% setting; the effective boost varies
    // according to the pixel's chroma and grayscale classification.
    float3 adjustedColor = AdjustSaturation(inputColor.rgb, 2.0);

    // Preserve the source alpha channel.
    return float4(adjustedColor, inputColor.a);
}
