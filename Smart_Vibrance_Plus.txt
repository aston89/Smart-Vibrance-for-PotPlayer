/*
    Smart Vibrance Plus v2.0 — PotPlayer

    This revision corrects the previous version's temporal-stabilization
    assumption: pPrev, read from c2, is a host-supplied constant that the
    pixel shader cannot update to retain the previous frame's state.
    Consequently, the original EMA formula cannot provide reliable
    frame-to-frame stabilization without an external history mechanism.

    v2.0 preserves the previous formula's zero-history response and adds
    a stateless low-light, low-chroma guard intended to reduce excessive
    chroma enhancement of compression noise.

    The new guard is a perceptual approximation, not a true temporal EMA.
    Genuine temporal stabilization requires persistent frame history
    managed by the host or an accessible previous-frame texture.
*/

sampler s0 : register(s0);
float4 p1 : register(c1);

#define PLUS_STABILITY_GUARD_STRENGTH 0.20
#define PLUS_CHROMA_RISK_START       0.0015
#define PLUS_CHROMA_RISK_END         0.0150
#define PLUS_SHADOW_RISK_START       0.0200
#define PLUS_SHADOW_RISK_END         0.1600

float sigmoid(float x)
{
    return 1.0 / (1.0 + exp(-x));
}

float smooth01(float x, float edge0, float edge1)
{
    float t = saturate((x - edge0) / max(edge1 - edge0, 1e-5));
    return t * t * (3.0 - 2.0 * t);
}

float3 AdjustColor(float3 color, float intensity, float pixelActivity)
{
    float luminance = dot(color, float3(0.2126, 0.7152, 0.0722));

    float3 chroma = color - luminance;
    float chromaEnergy = dot(chroma, chroma);
    float chromaMag = sqrt(chromaEnergy);

    float satRef  = lerp(0.35, 0.65, pixelActivity);
    float grayRef = lerp(0.006, 0.0015, pixelActivity);

    float normSat = saturate(chromaMag / satRef);
    float rolloff = 1.0 - normSat;

    float graySoft = sigmoid((grayRef - chromaEnergy) * 45.0);
    float response = lerp(rolloff, 1.0, graySoft);

    float gain = (intensity - 1.0) * response;

    return luminance + chroma + chroma * gain;
}

float4 main(float2 texCoord : TEXCOORD) : SV_TARGET
{
    float4 inputColor = tex2D(s0, texCoord);
    float3 color = inputColor.rgb;

    float luminance = dot(color, float3(0.2126, 0.7152, 0.0722));
    float3 chroma = color - luminance;
    float chromaEnergy = dot(chroma, chroma);

    float pixelActivityRaw = saturate(chromaEnergy * 10.0);
    float pixelActivity = pixelActivityRaw * (1.0 - exp(-0.08));

    float instantaneousChromaGuard =
        1.0 - saturate(chromaEnergy * (1.0 - exp(-0.25)) * 5.0);

    float lowChromaRisk = 1.0 - smooth01(
        chromaEnergy,
        PLUS_CHROMA_RISK_START,
        PLUS_CHROMA_RISK_END
    );

    float shadowRisk = 1.0 - smooth01(
        saturate(luminance),
        PLUS_SHADOW_RISK_START,
        PLUS_SHADOW_RISK_END
    );

    float lowLightChromaGuard =
        1.0 - PLUS_STABILITY_GUARD_STRENGTH *
        lowChromaRisk * shadowRisk;

    float intensity =
        2.0 *
        lerp(0.85, 1.0, pixelActivity) *
        instantaneousChromaGuard *
        lowLightChromaGuard;

    float3 outColor = AdjustColor(color, intensity, pixelActivity);

    return float4(outColor, inputColor.a);
}
