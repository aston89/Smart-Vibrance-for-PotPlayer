// =============================================================================
// Smart Vibrance PRO v2.0
// -----------------------------------------------------------------------------
// DirectX 9 / Pixel Shader 3.0
//
// This revision removes the previous-frame EMA assumption: pPrev was read from
// a host-supplied constant register, but the pixel shader could not update that
// register with measurements from the current frame. The former temporal
// stabilization therefore depended on state that was not reliably available.
//
// v2.0 replaces that path with stateless per-pixel activity and chroma
// calculations, preserving the former zero-history response, and adds a
// low-light / low-chroma guard to reduce excessive enhancement of compression
// noise in dark, weakly chromatic regions.
//
// The guard is a perceptual approximation, not temporal stabilization. Genuine
// frame-to-frame EMA requires persistent history supplied by the host.
//
// The PRO hue-response model remains intact: chroma enhancement is balanced
// continuously across the opponent-color plane, with restrained red/yellow
// response, more preserved green/blue response, and protection near neutral.
// =============================================================================

sampler s0 : register(s0);

// =============================================================================
// PRO RESPONSE MODEL
// =============================================================================
// Relative chroma response assigned to the four main opponent directions.
// Lower value = stronger restraint.
// Higher value = more of the original boost response is retained.
// These values form the control points of a continuous hue response rather
// than four discrete hue classes.
// Red is intentionally the most restrained region.
// Yellow is moderately restrained.
// Green and Blue remain closer to the original response.

#define PRO_RED_RESPONSE        0.50
#define PRO_YELLOW_RESPONSE     0.68
#define PRO_GREEN_RESPONSE      0.90
#define PRO_BLUE_RESPONSE       0.92

// Response used when chroma is too small for hue direction to be reliable.
// Near the neutral axis, hue becomes numerically unstable and perceptually ambiguous.
// Instead of forcing a hue decision, PRO falls back to this conservative neutral response.
#define PRO_NEUTRAL_RESPONSE    0.58

// -----------------------------------------------------------------------------
// Chroma range used for adaptive balancing.
//
// Below PRO_CHROMA_START:
//     strong adaptive protection.
//
// Between START and END:
//     protection gradually fades.
//
// Above PRO_CHROMA_END:
//     the original saturation rolloff dominates.
//
// This prevents low-chroma material from automatically receiving the same
// aggressive boost that would be useful on clearly defined colours.
#define PRO_CHROMA_START        0.075
#define PRO_CHROMA_END          0.26

// =============================================================================
// LOW-LIGHT / LOW-CHROMA GUARD
// =============================================================================
// Stateless protection against excessive chroma enhancement in dark regions
// with weak chroma. This is not frame-to-frame stabilization.

#define PRO_STABILITY_GUARD_STRENGTH 0.20
#define PRO_CHROMA_RISK_START        0.0015
#define PRO_CHROMA_RISK_END          0.0150
#define PRO_SHADOW_RISK_START        0.0200
#define PRO_SHADOW_RISK_END          0.1600

// =============================================================================
// BASIC UTILITY
// =============================================================================

// Smooth nonlinear response curve.
float sigmoid(float x)
{
    return 1.0 / (1.0 + exp(-x));
}


// Smooth 0..1 transition.
float smooth01(float x, float edge0, float edge1)
{
    float t = saturate((x - edge0) / max(edge1 - edge0, 1e-5));
    return t * t * (3.0 - 2.0 * t);
}


// =============================================================================
// CONTINUOUS HUE RESPONSE
// =============================================================================
//
// The chromatic plane is represented using two inexpensive opponent-style
// coordinates:
//
//     A = Red  <-> Green
//     B = Yellow <-> Blue
//
// These are not a full perceptual colour-space conversion. They are a
// lightweight directional representation suitable for a DX9 / PS_3_0 shader.
//
// The four main directional responses act as control points:
//
//                  Yellow
//                     |
//                     |
//          Green -----+----- Red
//                     |
//                     |
//                   Blue
//
// Mixed colours are automatically interpolated.
//
// Examples:
//
//     Orange      -> Red + Yellow
//     Magenta     -> Red + Blue
//     Cyan        -> Green + Blue
//     Yellow-Green-> Yellow + Green
//
// Therefore every hue receives a continuous response rather than entering a
// binary protected/unprotected category.
// =============================================================================

float HueResponse(
    float a,
    float b)
{
    // Positive A = red.
    float redWeight =
        max(a, 0.0);

    // Negative A = green.
    float greenWeight =
        max(-a, 0.0);

    // Positive B = yellow.
    float yellowWeight =
        max(b, 0.0);

    // Negative B = blue.
    float blueWeight =
        max(-b, 0.0);

    float total =
        redWeight +
        greenWeight +
        yellowWeight +
        blueWeight +
        1e-5;

    float response =
        (
            redWeight    * PRO_RED_RESPONSE +
            yellowWeight * PRO_YELLOW_RESPONSE +
            greenWeight  * PRO_GREEN_RESPONSE +
            blueWeight   * PRO_BLUE_RESPONSE
        ) / total;

    return response;
}

// =============================================================================
// COLOR ADJUSTMENT
// =============================================================================

float3 AdjustColor(
    float3 color,
    float intensity,
    float pixelActivity)
{

    // -------------------------------------------------------------------------
    // Perceptual luminance
    // The luminance component is preserved while chroma is adjusted
    // independently.
    // -------------------------------------------------------------------------

    float luminance =
        dot(
            color,
            float3(
                0.2126,
                0.7152,
                0.0722
            )
        );

    // -------------------------------------------------------------------------
    // Opponent-style chroma coordinates
    //
    // A:
    //     positive = red
    //     negative = green
    //
    // B:
    //     positive = yellow
    //     negative = blue
    //
    // These coordinates provide a cheap directional chroma representation
    // without requiring a full perceptual colour-space conversion.
    // -------------------------------------------------------------------------

    float A =
        color.r -
        color.g;

    float B =
        (color.r + color.g) * 0.5 -
        color.b;

    // -------------------------------------------------------------------------
    // Chroma magnitude
    // Measures how far the colour lies from the neutral axis in the reduced
    // opponent plane.
    // -------------------------------------------------------------------------

    float chromaEnergy =
        A * A +
        B * B;

    float chromaMag =
        sqrt(chromaEnergy);

    // -------------------------------------------------------------------------
    // Pixel-adaptive parameters
    // The saturation reference expands with the current pixel's activity,
    // allowing the response curve to adapt instead of using one fixed
    // saturation threshold for every pixel.
    // -------------------------------------------------------------------------

    float satRef =
        lerp(
            0.35,
            0.65,
            pixelActivity
        );

    float grayRef =
        lerp(
            0.006,
            0.0015,
            pixelActivity
        );

    // -------------------------------------------------------------------------
    // Nonlinear chroma rolloff
    // Weak colours receive more available boost.
    // Increasingly strong colours progressively consume that boost budget.
    // -------------------------------------------------------------------------

    float normSat =
        saturate(
            chromaMag / satRef
        );

    float rolloff =
        1.0 - normSat;


    // -------------------------------------------------------------------------
    // Near-neutral protection
    // Colours close to the neutral axis receive a controlled fallback response
    // rather than being allowed to inherit the maximum boost.
    // -------------------------------------------------------------------------

    float graySoft =
        sigmoid(
            (grayRef - chromaEnergy) * 45.0
        );

    float response =
        lerp(
            rolloff,
            1.0,
            graySoft
        );

    // =========================================================================
    // CONTINUOUS HUE BALANCING
    // =========================================================================
    // Hue direction is useful only when enough chroma is present to make that
    // direction meaningful. Very weak chroma therefore blends toward the
    // neutral fallback response.
    // =========================================================================

    float hueConfidence =
        saturate(
            chromaMag / 0.020
        );

    float hueResponse =
        HueResponse(
            A,
            B
        );

    float balancedResponse =
        lerp(
            PRO_NEUTRAL_RESPONSE,
            hueResponse,
            hueConfidence
        );

    // =========================================================================
    // CHROMA-DEPENDENT BOOST BALANCING
    // =========================================================================
    // The directional response is strongest where chroma is weak or moderate.
    // As chroma increases, the balancing influence progressively disappears and
    // the original nonlinear saturation rolloff becomes dominant again.
    // This prevents the filter from treating every low-saturation colour as an
    // opportunity for maximum enhancement.
    // =========================================================================

    float chromaPosition =
        saturate(
            (chromaMag - PRO_CHROMA_START) /
            max(
                PRO_CHROMA_END -
                PRO_CHROMA_START,
                1e-5
            )
        );

    float chromaFade =
        1.0 -
        chromaPosition;

    // Smooth the transition to avoid visible response boundaries.
    chromaFade =
        chromaFade *
        chromaFade *
        (3.0 - 2.0 * chromaFade);

    // -------------------------------------------------------------------------
    // Calculate how far the directional response is from full boost.
    // balancedResponse = 1.0 (no additional restriction)
    // balancedResponse = 0.50 (strongly moderated chroma response)
    // -------------------------------------------------------------------------

    float protection =
        chromaFade *
        (
            1.0 -
            balancedResponse
        );


    // Apply the directional balance to the original adaptive response.
    response *=
        1.0 -
        protection;

    // -------------------------------------------------------------------------
    // Final adaptive chroma gain
    // -------------------------------------------------------------------------

    float gain =
        (intensity - 1.0) *
        response;


    // Reconstruct RGB from the original luminance and adjusted chroma.
    // Because chroma is scaled rather than independently modifying RGB channels, the hue direction itself is preserved while its intensity is controlled adaptively.
    return
        luminance +
        (color - luminance) +
        (color - luminance) * gain;
}

// =============================================================================
// MAIN
// =============================================================================

float4 main(float2 texCoord : TEXCOORD0) : COLOR
{
    // -------------------------------------------------------------------------
    // Input frame
    // -------------------------------------------------------------------------

    float4 inputColor =
        tex2D(
            s0,
            texCoord
        );

    float3 color =
        inputColor.rgb;

    // -------------------------------------------------------------------------
    // Current luminance
    // -------------------------------------------------------------------------

    float luminance =
        dot(
            color,
            float3(
                0.2126,
                0.7152,
                0.0722
            )
        );


    // -------------------------------------------------------------------------
    // Current chroma energy
    // -------------------------------------------------------------------------

    float3 chroma =
        color -
        luminance;

    float chromaEnergy =
        dot(
            chroma,
            chroma
        );

    // =========================================================================
    // PIXEL ACTIVITY
    // =========================================================================
    // Uses the current pixel only. The coefficient below preserves the former
    // EMA's response when its unavailable previous-state input was zero; it is
    // not a temporally smoothed or frame-wide scene measurement.
    // =========================================================================

    float pixelActivityRaw =
        saturate(
            chromaEnergy * 10.0
        );

    float pixelActivity =
        pixelActivityRaw *
        (1.0 - exp(-0.08));


    // =========================================================================
    // INSTANTANEOUS CHROMA GUARD
    // =========================================================================
    // Replaces the former microActivity EMA using the equivalent zero-history
    // weighting. It reacts only to the current pixel and retains no frame state.
    // =========================================================================

    float instantaneousChromaGuard =
        1.0 -
        saturate(
            chromaEnergy *
            (1.0 - exp(-0.25)) *
            5.0
        );

    // =========================================================================
    // LOW-LIGHT / LOW-CHROMA GUARD
    // =========================================================================
    // Further reduces chroma enhancement in dark regions with weak chroma,
    // where compression noise can become conspicuous after enhancement.
    // This is a stateless perceptual approximation, not frame-to-frame analysis.
    // =========================================================================

    float lowChromaRisk =
        1.0 - smooth01(
            chromaEnergy,
            PRO_CHROMA_RISK_START,
            PRO_CHROMA_RISK_END
        );

    float shadowRisk =
        1.0 - smooth01(
            saturate(luminance),
            PRO_SHADOW_RISK_START,
            PRO_SHADOW_RISK_END
        );

    float lowLightChromaGuard =
        1.0 -
        PRO_STABILITY_GUARD_STRENGTH *
        lowChromaRisk *
        shadowRisk;

    // =========================================================================
    // ADAPTIVE INTENSITY
    // =========================================================================
    // The former zero-history response is retained, then modulated by the
    // low-light / low-chroma guard. No previous-frame state is assumed.
    // =========================================================================

    float intensity =
        2.0 *
        lerp(
            0.85,
            1.0,
            pixelActivity
        ) *
        instantaneousChromaGuard *
        lowLightChromaGuard;

    // -------------------------------------------------------------------------
    // Final adaptive colour adjustment
    // -------------------------------------------------------------------------

    float3 outColor =
        AdjustColor(
            color,
            intensity,
            pixelActivity
        );

    return float4(
        outColor,
        inputColor.a
    );
}
